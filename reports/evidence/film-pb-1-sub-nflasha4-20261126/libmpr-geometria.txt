## 5. CtrlInfoBlockNum: heuristica por numero total de bloques (0x2A9C4A-0x2A9C7E)

   002a9c70  movs r3, #2
   002a9c74  cmp r1, #0x50
   002a9c78  movs r3, #6
   002a9c7c  movs r3, #8
